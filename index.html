RxJS search autocomplete

This search field waits 300 ms after typing stops before looking up users. It searches when the text has at least two characters and clears the results for shorter queries. Each new value passed to switchMap cancels the previous request. If a request fails, the results are cleared and the search keeps working. 

interface User {
  id: string;
  name: string;
}

readonly search = new FormControl<string>('', {
  nonNullable: true,
});

private searchUsers(query: string): Observable<User[]> {
  return this.api.search(query).pipe(
    catchError(() => of<User[]>([]))
  );
}

readonly results$: Observable<User[]> = this.search.valueChanges.pipe(
  map(value => value.trim()),
  distinctUntilChanged(),
  debounceTime(300),
  switchMap(query =>
    query.length >= 2
      ? this.searchUsers(query)
      : of<User[]>([])
  )
);
