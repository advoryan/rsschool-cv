Code Example

RxJS search autocomplete

An Angular form control with a 300 ms debounce. Queries with at least two characters reach switchMap, which replaces the previous API subscription. 

private readonly destroyRef = inject(DestroyRef);
readonly search = new FormControl('', { nonNullable: true });

readonly results$ = this.search.valueChanges.pipe(
  map(value => value.trim()),
  debounceTime(300),
  distinctUntilChanged(),
  filter(value => value.length >= 2),
  switchMap(query => this.api.search(query)),
  catchError(error => {
    console.error(error);
    return of([]);
  }),
  takeUntilDestroyed(this.destroyRef),
);
